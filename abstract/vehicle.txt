abstract class Vehicle {

    abstract void start();
}

class Car extends Vehicle {

    void start() {
        System.out.println("Car Started");
    }
}

class Bike extends Vehicle {

    void start() {
        System.out.println("Bike Started");
    }
}

public class Test {
    public static void main(String[] args) {

        Vehicle v1 = new Car();
        v1.start();

        Vehicle v2 = new Bike();
        v2.start();
    }
}
