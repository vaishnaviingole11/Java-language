package p1;

class Bill {

    void calculateBill() {
        System.out.println("Calculating Bill...");
    }

    void display() {
        System.out.println("Bill Details");
    }
}

class ElectricityBill extends Bill {

    int customerId;
    String customerName;
    int unit;
    static double ratePerUnit = 9;
    double totalBillAmount;

    ElectricityBill(int customerId, String customerName, int unit) {
        this.customerId = customerId;
        this.customerName = customerName;
        this.unit = unit;
    }

    @Override
    void calculateBill() {
        totalBillAmount = unit * ratePerUnit;
    }

    @Override
    void display() {
        System.out.println("Customer Id : " + customerId);
        System.out.println("Customer Name : " + customerName);
        System.out.println("Unit : " + unit);
        System.out.println("Rate Per Unit : " + ratePerUnit);
        System.out.println("Total Bill Amount : " + totalBillAmount);
        System.out.println("-------------------------");
    }
}

public class TestElectricity {

    public static void main(String[] args) {

        // Upcasting
        Bill b1 = new ElectricityBill(1, "Vaishnavi", 50);
        Bill b2 = new ElectricityBill(2, "Anjali", 60);
        Bill b3 = new ElectricityBill(3, "Ram", 34);

        b1.calculateBill();
        b2.calculateBill();
        b3.calculateBill();

        b1.display();
        b2.display();
        b3.display();

        ElectricityBill.ratePerUnit = 10;

        System.out.println("After Changing Rate Per Unit");
        System.out.println();

        b1.calculateBill();
        b2.calculateBill();
        b3.calculateBill();

        b1.display();
        b2.display();
        b3.display();
    }
}
