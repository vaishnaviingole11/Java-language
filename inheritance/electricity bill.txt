

class ElectricityBill{
	int customerId;
	String  customerName;
	int unit;
	static double rateperunit=9;
	double TotalBillAmount;

	ElectricityBill(int customerId, String customerName, int unit) {
	this.customerId = customerId;
	this.customerName = customerName;
	this.unit = unit;
   }
	int getCustomerId() {
		return customerId;
	}
	void setCustomerId(int customerId) {
		this.customerId = customerId;
	}
	String getCustomerName() {
		return customerName;
	}
	void setCustomerName(String customerName) {
		this.customerName = customerName;
	}
	int getUnit() {
		return unit;
	}
	void setUnit(int unit) {
		this.unit = unit;
	}
	static double getRateperunit() {
		return rateperunit;
	}
	static void setRateperunit(double rateperunit) {
		ElectricityBill.rateperunit = rateperunit;
	}
	double getTotalBillAmount() {
		return TotalBillAmount;
	}
	void setTotalBillAmount(double totalBillAmount) {
		TotalBillAmount = totalBillAmount;
	}


void display() {
	            System.out.println("customer id is :"+this.customerId);
				System.out.println("customer name is:"+this.customerName);
				System.out.println("unit is:"+this.unit);
				System.out.println("rateperunit is:"+this.rateperunit);
				System.out.println("total Bill Amount:"+(TotalBillAmount=unit*rateperunit));

}
}
